data/javascript/01-event-loop-async.md
# Event Loop, Microtasks & Async JavaScript

## Slide 1: The Single-Threaded Runtime Model
JavaScript executes code on a single main thread:
- **Call Stack:** Executes synchronous functions (LIFO).
- **Task / Macrotask Queue:** `setTimeout`, `setInterval`.
- **Microtask Queue:** `Promise.then`, `queueMicrotask`.

## Slide 2: Microtask Queue vs. Macrotask Queue
After the Call Stack empties, the Event Loop drains
**100% of the Microtask Queue** before taking the
next Macrotask:
`Sync Code -> All Microtasks -> Next Macrotask`
data/javascript/02-closures-scopes.md
# Closures, Lexical Scope & Memory

## Slide 1: Lexical Scoping & Closure Definition
A **closure** is a function bundled together with
references to its surrounding lexical environment.

## Slide 2: Practical Closure Patterns & Loop Trap
- Using `var` in a `for` loop shares ONE binding
  across all `setTimeout` callbacks.
- Using `let` creates a fresh block-scoped binding
  per loop iteration.

Loading Area Content: src/shared/content-loader.js

getAreaArg() reads the area parameter from CLI (node index.js javascript), and loadAreaContent(area, customContent) loads either data/[area]/*.md or a passed customContent string.

src/shared/content-loader.js (Part 1 — Area & File Loader)
import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
export const DATA_DIR = path.resolve(__dirname, '../../data');

export function getAreaArg(defaultArea = 'javascript') {
  const args = process.argv.slice(2).filter((a) => a !== '--demo');
  for (const arg of args) {
    if (arg.startsWith('--area=')) return arg.split('=')[1];
    if (!arg.startsWith('-')) return arg;
  }
  return process.env.AREA || defaultArea;
}

export function loadAreaContent(area = 'javascript', customContent = '') {
  const areaPath = path.join(DATA_DIR, area);
  const files = [];
  const chunks = [];

  if (customContent && customContent.trim()) {
    files.push({ file: 'custom-parameter.md', content: customContent.trim() });
    chunks.push(...chunkMarkdownContent(area, 'custom-parameter.md', customContent));
  } else {
    for (const entry of fs.readdirSync(areaPath, { withFileTypes: true })) {
      if (!entry.isFile() || !/\.(md|txt)$/i.test(entry.name)) continue;
      const raw = fs.readFileSync(path.join(areaPath, entry.name), 'utf-8');
      files.push({ file: entry.name, content: raw.trim() });
      chunks.push(...chunkMarkdownContent(area, entry.name, raw));
    }
  }
  return { area, files, chunks };
}
Slide 06 / 16 • Shared Helper 2

Chunking & Searching Area Notes: src/shared/content-loader.js

searchAreaContent({ area, query, topK, customContent }) ranks chunks by keyword matches in the section body and heading.

src/shared/content-loader.js (Part 2 — Keyword Search Engine)
export function searchAreaContent({ area = 'javascript', query = '', topK = 3, customContent = '' } = {}) {
  const { area: resolvedArea, chunks } = loadAreaContent(area, customContent);
  if (!query || !query.trim()) {
    return { area: resolvedArea, query, results: chunks.slice(0, topK) };
  }

  const terms = query
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, ' ')
    .split(/\s+/)
    .filter((t) => t.length >= 2);

  const scored = chunks.map((chunk) => {
    const haystack = `${chunk.docTitle} ${chunk.heading} ${chunk.content}`.toLowerCase();
    let score = 0;
    for (const term of terms) {
      if (haystack.includes(term)) {
        score += 1;
        if (chunk.heading.toLowerCase().includes(term)) score += 1.5;
      }
    }
    return { ...chunk, score: Number(score.toFixed(2)) };
  });

  scored.sort((a, b) => b.score - a.score);
  return { area: resolvedArea, query, results: scored.slice(0, topK) };
}
Slide 07 / 16 • Shared Helper 3

Google ADK Local Ollama Adapter: src/shared/ollama-llm.js

Extends BaseLlm from @google/adk and calls Ollama's /api/chat endpoint via native fetch(), translating ADK tool declarations and function calls automatically.

src/shared/ollama-llm.js
import { BaseLlm, InMemoryRunner } from '@google/adk';

export async function ollamaChat({ model = 'gemma4:e2b', messages, tools, format, options }) {
  const res = await fetch('http://127.0.0.1:11434/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ model, messages, tools, format, options, stream: false }),
  });
  return res.json();
}

export class OllamaGemmaLlm extends BaseLlm {
  constructor({ model = 'gemma4:e2b', temperature = 0.2 } = {}) {
    super({ model });
    this.model = model;
    this.temperature = temperature;
  }

  async *generateContentAsync(llmRequest) {
    // 1. Convert llmRequest.contents & llmRequest.config.tools to Ollama format
    const response = await ollamaChat({
      model: this.model,
      messages,
      tools: tools.length > 0 ? tools : undefined,
      options: { temperature: this.temperature },
    });

    const msg = response.message || {};
    if (msg.tool_calls?.length) {
      yield {
        content: {
          role: 'model',
          parts: msg.tool_calls.map((tc) => ({
            functionCall: { name: tc.function.name, args: tc.function.arguments },
          })),
        },
      };
      return;
    }
    yield { content: { role: 'model', parts: [{ text: msg.content || '' }] } };
  }
}

export async function generateExamQuestion({ area = 'javascript', topic = '', customContent = '' } = {}) {
  const rag = searchAreaContent({ area, query: topic || area, topK: 4, customContent });
  const contextText = rag.results
    .map((r) => `[Source: ${r.file} - ${r.heading}]\n${r.excerpt}`)
    .join('\n\n');

  const response = await ollamaChat({
    model: DEFAULT_MODEL,
    messages: [
      {
        role: 'system',
        content: 'You are an exam designer. Return valid JSON matching the requested schema grounded strictly in the provided notes.',
      },
      {
        role: 'user',
        content: `Create a 20-point technical exam question for area "${area}"...\n\n${contextText}`,
      },
    ],
    format: EXAM_JSON_SCHEMA,
    options: { temperature: 0.2 },
  });

  const parsed = JSON.parse(response.message.content);
  parsed.area = area;
  parsed.totalPoints = 20;
  return ExamQuestionSchema.parse(parsed);
}

export async function evaluateStudentAnswer({ area = 'javascript', exam, studentAnswer, customContent = '' }) {
  const rag = searchAreaContent({ area, query: exam.questionPrompt, topK: 3, customContent });
  const referenceNotes = rag.results.map((r) => `[${r.file} - ${r.heading}]\n${r.excerpt}`).join('\n\n');

  const response = await ollamaChat({
    model: DEFAULT_MODEL,
    messages: [
      { role: 'system', content: 'You are a fair, deterministic technical exam grader. Return only valid JSON.' },
      {
        role: 'user',
        content: `EXAM QUESTION:\n${exam.questionPrompt}\n\nRUBRIC:\n${JSON.stringify(exam.rubric)}\n\nREFERENCE NOTES (data/${area}/):\n${referenceNotes}\n\nSTUDENT ANSWER:\n${studentAnswer}`,
      },
    ],
    format: EVALUATION_JSON_SCHEMA,
    options: { temperature: 0.1 },
  });

  const parsed = JSON.parse(response.message.content);
  parsed.totalAwarded = (parsed.criterionScores || []).reduce((s, c) => s + Number(c.awardedPoints || 0), 0);
  parsed.totalPossible = (exam.rubric || []).reduce((s, r) => s + r.maxPoints, 0) || 20;
  parsed.percentage = Math.round((parsed.totalAwarded / parsed.totalPossible) * 100);
  return EvaluationSchema.parse(parsed);
}

export function MarkdownBox({
  label,
  badge,
  value,
  onChange,
  placeholder = 'Write Markdown here...',
  rows = 6,
  defaultMode = 'preview',
  readOnly = false,
}) {
  const [mode, setMode] = useState(defaultMode);
  const effectiveMode = readOnly ? 'preview' : mode;

  return (
    <div className="md-box-wrapper">
      <div className="md-box-toolbar">
        <span>{label}</span>
        {!readOnly && (
          <div className="md-mode-tabs">
            {['preview', 'split', 'edit'].map((tab) => (
              <button key={tab} onClick={() => setMode(tab)} className={effectiveMode === tab ? 'active' : ''}>
                {tab === 'preview' ? 'Markdown Preview' : tab === 'split' ? 'Split' : 'Edit MD'}
              </button>
            ))}
          </div>
        )}
      </div>
      {effectiveMode === 'preview' && <MarkdownPreview content={value} emptyPlaceholder={placeholder} />}
      {effectiveMode === 'edit' && <textarea rows={rows} value={value} onChange={(e) => onChange(e.target.value)} />}
    </div>
  );
}
